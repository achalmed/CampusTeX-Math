	\section*{Trabajando en clase}
	\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.4cm]
		\item Reducir:
		      $$
			      A =  \frac{(x^7)^m(x^3)^m}{x^{10m}}
		      $$

		\item Reducir:
		      $$
			      \frac{(4m^2 \cdot n^3) \cdot (2mn \cdot 3m^2 \cdot n)}{(2mn)^3 \cdot (m^2n)}
		      $$
		\item Reducir:
		      $$
			      M = (a^3)^2*a^{-3^2}*a^{(+3)^2}*a^{(3^2)^2}
		      $$

		\item Calcular:
		      $$
			      M = \frac{\frac{1}{2} \cdot \left( \frac{1}{3} \right)^2 \cdot \left( \frac{1}{6} \right)^3}{\frac{1}{2} \cdot \left( \frac{1}{3} \right)^3 \cdot \left( \frac{1}{6} \right)^2}
		      $$

		\item Calcular:
		      $$
			      \left( \frac{2}{3} \right)^2 + \left( \frac{1}{3} \right)^2 + \left( \frac{2}{3} \right)^2 \cdot \left( \frac{1}{3} \right)^2 = 0
		      $$

		\item Simplificar:
		      $$
			      R = \frac{2^5 \cdot 3^2 \cdot 5^4 \cdot 7^2}{2^3 \cdot 3^4 \cdot 5^2 \cdot 7^4}
		      $$

		\item Calcular:
		      $$
			      A = \frac{2(3 + 2)^2 - 3(2^2 - 1)^2}{(3 + 1)^3 - 3^2} = 30
		      $$

		\item Simplificar:
		      $$
			      A = \frac{3(4 + 3)^2 - 3(2^2 + 1)^2}{3(4 + 1)^3 - 3^2} = 3
		      $$

		\item Se factoriza, tanto en el numerador como en el denominador, la expresión:
		      $$
			      \frac{3(a^3 - b^3) - 3(a - b)^3}{3(a^3 + b^3) - 3(a + b)^3}
		      $$

		\item Calcular:
		      $$
			      \frac{3^5 \cdot 5^2 \cdot 4^3 \cdot 5^5}{3^2 \cdot 5^5 \cdot 4^5 \cdot 5^2}
		      $$

		\item Resolver:
		      $$
			      \frac{4(4 - 4) - 4(8 + 8) + 8 - 8}{(4 - 4)^2}
		      $$

		\item Luego de resolver:
		      $$
			      2^3 \cdot 4^8 \cdot 8^3 \cdot 16^5
		      $$

		\item Da como respuesta el valor de $x$:
		      $$
			      2^{x} \cdot 2^{x+1} \cdot 2^{x+2} = 2^{x+8} \cdot 4^{x+2}
		      $$

		\item Al tener productos de bases iguales se tiene:
		      $$
			      \begin{cases}
				      2^{x} \cdot 2^{x+1} \cdot 2^{x+2} = 2^{x+8} \cdot 4^{x+2} \\
				      \ldots x \text{ sea } 8 \text{ para } x + 4 = 12          \\
				      \text{Luego} x = 2
			      \end{cases}
		      $$

		\item La respuesta $x^3 + x$ da como respuesta:
		      $$
			      3^{x+1} \cdot 8^{x+1} \cdot 245 \cdot 2^{x+5}
		      $$

		\item Determinar el valor de:
		      $$
			      \left( \frac{x^3 + y^3}{x^3 - y^3} \right)
		      $$
