% Caja de datos del problema (MRUA, etc.)
\newcommand{\datosprob}[1]{%
  \begin{tcolorbox}[
    enhanced, colback=GrisClaro, colframe=AzulMedio,
    arc=2pt, boxrule=0.6pt,
    fonttitle=\sffamily\bfseries\small,
    title={Datos del problema},
    left=6pt, right=6pt, top=4pt, bottom=4pt
  ]
  \begin{multicols}{2}
  #1
  \end{multicols}
  \end{tcolorbox}
}
